  from utils.executor_factory import ExecutorFactory

  async def grade_code():
      # Create executor
      executor = await ExecutorFactory.create_executor()
      
      # Execute securely
      result = await executor.execute(student_code, test_code)
      
      # Get results
      print(f"Score: {result.score}/{result.max_score}")
      
      # Cleanup
      await executor.cleanup()

┌──────────────────────────────────────────────────────────────────────────┐
│ 🛠️ TROUBLESHOOTING                                                       │
└──────────────────────────────────────────────────────────────────────────┘

  Common Issues:

    ❌ "Docker client not available"
       → Ensure Docker Desktop is running

    ❌ "Image not found"
       → Build image: cd docker && ./build.ps1

    ❌ "Permission denied" (Linux)
       → Add to docker group: sudo usermod -aG docker $USER

    ❌ "Named pipe error" (Windows)
       → Set DOCKER_HOST=tcp://localhost:2375

    ❌ Timeouts on every execution
       → Increase DEFAULT_TIMEOUT in .env

  See docker/README.md for detailed troubleshooting.
